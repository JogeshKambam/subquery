1. create employees table


create table employees (
    emp_id int primary key,
    first_name varchar(50),
    last_name varchar(50),
    salary decimal(10,2),
    department_id int
);


2. create salary_audit table

create table salary_audit (
    audit_id int  primary key,
    emp_id int,
    old_salary decimal(10,2),
    new_salary decimal(10,2),
    operation varchar(10),
    changed_at timestamp,
    changed_by varchar(100)
);


3. before insert trigger

Here this trigger records the new employee's salary before inserting the employee into the table.


delimiter $$

create trigger before_employee_insert
before insert on employees
for each row
begin
    insert into salary_audit
    values (
        null,
        new.emp_id,
        null,
        new.salary,
        'insert',
        current_timestamp,
        current_user()
    );
end $$

delimiter ;


4. after update trigger

This trigger records the old and new salaries only when the salary changes.


delimiter $$

create trigger after_employee_update
after update on employees
for each row
begin
    if old.salary <> new.salary then
        insert into salary_audit
        values (
            null,
            new.emp_id,
            old.salary,
            new.salary,
            'update',
            current_timestamp,
            current_user()
        );
    end if;
end $$

delimiter ;

 5. after delete trigger

This trigger records the employee's old salary when the employee is deleted or removed .


delimiter $$

create trigger after_employee_delete
after delete on employees
for each row
begin
    insert into salary_audit
    values (
        null,
        old.emp_id,
        old.salary,
        null,
        'delete',
        current_timestamp,
        current_user()
    );
end $$

delimiter ;

 7. update an employee's salary

Here, employee 1 salary changes from 25000 to 28000 using update .


update employees
set salary = 28000
where emp_id = 1;


 8. delete an employee

Here, employee 3 is deleted from the employees table.


delete from employees
where emp_id = 3;
